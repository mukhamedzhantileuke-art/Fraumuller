/**
 * Validates password strength according to security checklist
 * @param {string} password - Raw password input
 * @returns {Object} Validation states for each rule
 */
function validatePasswordRules(password) {
    const minLength = 8;
    const hasUpper = /[A-ZА-ЯЁ]/.test(password);
    const hasLower = /[a-zа-яё]/.test(password);
    const hasDigit = /[0-9]/.test(password);
    const hasSpecial = /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password);
    const isLengthValid = password.length >= minLength;

    const allPassed = isLengthValid && hasUpper && hasLower && hasDigit && hasSpecial;

    return {
        isLengthValid,
        hasUpper,
        hasLower,
        hasDigit,
        hasSpecial,
        allPassed
    };
}

                updateItem('#rule-special', valResult.hasSpecial, 'Специальный символ (!@#$%^&*)');
                updateItem('#rule-match', isMatch, 'Пароли совпадают');

                return valResult.allPassed && isMatch;

                 * 2. Browser scripts cannot check database uniqueness (e.g., already registered emails).
                 * 3. Malicious payloads could compromise the backend if the server blindly trusts client data.

                if (!isValid) {

                $passwordInput.removeClass('is-invalid').addClass('is-valid');
                $confirmInput.removeClass('is-invalid').addClass('is-valid');

                const $successDiv = $('<div>', {
                    class: 'alert alert-success py-2 small text-start'
                }).hide();

                const $strong = $('<strong>', { class: 'd-block mb-1', text: 'Демонстрационный режим:' });
                const $pDesc = $('<p>', {

                    text: 'Проверка данных успешно пройдена на стороне клиента. В учебных целях данные не передаются на сервер.'
                });

                $successDiv.append($strong).append($pDesc);
                $messageContainer.append($successDiv);

                // Effect with duration and callback
                $successDiv.fadeIn(400, function () {
                    $registerForm.trigger('reset');
                    updateChecklist();
                });
            });
        }
    }
}

/**
 * Reusable function 1: Formats number into KZT currency string
 * @param {number} amount - Numeric price
 * @returns {string} Formatted string with currency sign
 */
function formatCurrency(amount) {
    return amount.toLocaleString('ru-RU') + ' ₸';
}

/**
 * Reusable function 2: Calculates discounted price with percentage
 * @param {number} originalPrice - Base price
 * @param {number} discountPercent - Discount percent (e.g. 15)
 * @returns {number} Final discounted price rounded to nearest 50
 */
function calculateDiscountPrice(originalPrice, discountPercent) {
    const rawDiscounted = originalPrice * (1 - discountPercent / 100);
    return Math.round(rawDiscounted / 50) * 50;
}

        $(this).closest('.card').find('.card-title').addClass('text-danger');

/**
 * Global Cart State for Menu
 */
const MenuCart = {
    items: {},

    load() {
        try {
            const saved = localStorage.getItem('fraumuller_cart');
            if (saved) {
                this.items = JSON.parse(saved);
            }
        } catch (e) {
            this.items = {};
        }
    },

    save() {
        try {
            localStorage.setItem('fraumuller_cart', JSON.stringify(this.items));
        } catch (e) {
            // LocalStorage fallback
        }
    },

    addItem(name, price, image) {
        if (!this.items[name]) {
            this.items[name] = { name: name, price: price, qty: 1, image: image || '' };
        } else {
            this.items[name].qty += 1;
        }
        this.save();
    },

    removeItem(name) {
        if (this.items[name]) {
            this.items[name].qty -= 1;
            if (this.items[name].qty <= 0) {
                delete this.items[name];
            }
            this.save();
        }
    },

    clear() {
        this.items = {};
        this.save();
    },

    getTotalPrice() {
        let total = 0;
        $.each(this.items, function (key, item) {
            total += item.price * item.qty;
        });
        return total;
    },

    getTotalCount() {
        let count = 0;
        $.each(this.items, function (key, item) {
            count += item.qty;
        });
        return count;
    }
};

function initMenuSearch() {
    const $searchInput = $('#menu-search-input');
    const $clearBtn = $('#menu-search-clear');
    const $resultsInfo = $('#search-results-info');

    if (!$searchInput.length) {
        return;
    }

    function doSearch() {
        const query = $.trim($searchInput.val()).toLowerCase();
        const $rows = $('tbody tr[data-dish-name]');
        let matchCount = 0;

        $rows.each(function () {
            const $row = $(this);
            const name = $row.attr('data-dish-name') || '';
            const desc = $row.attr('data-dish-desc') || '';

            if (query === '' || name.indexOf(query) !== -1 || desc.indexOf(query) !== -1) {
                $row.removeClass('d-none');
                matchCount++;
            } else {
                $row.addClass('d-none');
            }
        });

        // Hide empty menu sections using :not() filter
        $('section.card').each(function () {
            const $sec = $(this);
            const $visibleRows = $sec.find('tbody tr:not(.d-none)');
            if ($visibleRows.length === 0 && query !== '') {
                $sec.addClass('d-none');
            } else {
                $sec.removeClass('d-none');
            }
        });

        // Update info text
        if ($resultsInfo.length) {
            if (query !== '') {
                $resultsInfo.removeClass('d-none').text('Найдено позиций: ' + matchCount);
            } else {
                $resultsInfo.addClass('d-none');
            }
        }
    }

    $searchInput.on('input', doSearch);

    if ($clearBtn.length) {
        $clearBtn.on('click', function () {
            $searchInput.val('');
            doSearch();
            $searchInput.focus();
        });
    }
}

'use strict';

/**
 * Initializes review filters, sorting controls, and dynamic submission using jQuery
 */
function initFeedbackInteractivity() {
    const $reviewsContainer = $('#reviews-container');
    const $starFiltersContainer = $('#star-filters');
    const $sortDateBtn = $('#sort-date');
    const $sortRatingBtn = $('#sort-rating');
    const $sortUsefulnessBtn = $('#sort-usefulness');
    const $feedbackForm = $('#feedback-form');
    const $feedbackMessageContainer = $('#feedback-message-container');

    if (!$reviewsContainer.length) {
        return;
    }

    let activeFilter = 'all';
    let currentSort = 'date';
    let sortDirection = 'desc';

    // 1. Star Filters using jQuery traversal and $(this)
    if ($starFiltersContainer.length) {
        const $filterButtons = $starFiltersContainer.find('.filter-btn');

        $filterButtons.css('cursor', 'pointer');
        $filterButtons.on('click', function () {
            activeFilter = $(this).attr('data-filter') || 'all';

            $filterButtons.removeClass('active-filter text-dark border-bottom border-2 border-dark pb-1')
                          .addClass('text-muted');

            $(this).removeClass('text-muted')
                   .addClass('active-filter text-dark border-bottom border-2 border-dark pb-1');

            applyFiltersAndSort();
        });
    }

    // 2. Sorting Controls
    function updateSortButtonsUI($activeBtn, labelText) {
        $.each([$sortDateBtn, $sortRatingBtn, $sortUsefulnessBtn], function (idx, $btn) {
            if ($btn && $btn.length) {
                $btn.removeClass('text-dark fw-bold border-bottom border-dark')
                    .addClass('text-muted');
            }
        });

        if ($activeBtn && $activeBtn.length) {
            const arrow = sortDirection === 'desc' ? ' ↓' : ' ↑';
            $activeBtn.removeClass('text-muted')
                      .addClass('text-dark fw-bold border-bottom border-dark')
                      .text(labelText + arrow);
        }
    }

    if ($sortDateBtn.length) {
        $sortDateBtn.on('click', function () {
            if (currentSort === 'date') {
                sortDirection = sortDirection === 'desc' ? 'asc' : 'desc';
            } else {
                currentSort = 'date';
                sortDirection = 'desc';
            }
            updateSortButtonsUI($sortDateBtn, 'Дате');
            applyFiltersAndSort();
        });
    }

    if ($sortRatingBtn.length) {
        $sortRatingBtn.on('click', function () {
            if (currentSort === 'rating') {
                sortDirection = sortDirection === 'desc' ? 'asc' : 'desc';
            } else {
                currentSort = 'rating';
                sortDirection = 'desc';
            }
            updateSortButtonsUI($sortRatingBtn, 'Рейтингу');
            applyFiltersAndSort();
        });
    }

    if ($sortUsefulnessBtn.length) {
        $sortUsefulnessBtn.on('click', function () {
            if (currentSort === 'usefulness') {
                sortDirection = sortDirection === 'desc' ? 'asc' : 'desc';
            } else {
                currentSort = 'usefulness';
                sortDirection = 'desc';
            }
            updateSortButtonsUI($sortUsefulnessBtn, 'Полезности');
            applyFiltersAndSort();
        });
    }

    /**
     * Reorders and toggles visibility of review cards based on active filter and sort state
     */
    function applyFiltersAndSort() {
        const $cards = $reviewsContainer.find('.review-card').toArray();
        let visibleCount = 0;

        // Sort cards
        $cards.sort(function (a, b) {
            const $a = $(a);
            const $b = $(b);

            if (currentSort === 'date') {
                const dateA = $a.attr('data-date') || '2026-01-01';
                const dateB = $b.attr('data-date') || '2026-01-01';
                return sortDirection === 'desc' ? dateB.localeCompare(dateA) : dateA.localeCompare(dateB);
            }

            let valA = 0;
            let valB = 0;
            if (currentSort === 'rating') {
                valA = parseInt($a.attr('data-rating') || '0', 10);
                valB = parseInt($b.attr('data-rating') || '0', 10);
            } else if (currentSort === 'usefulness') {
                valA = parseInt($a.attr('data-usefulness') || '0', 10);
                valB = parseInt($b.attr('data-usefulness') || '0', 10);
            }

            return sortDirection === 'desc' ? valB - valA : valA - valB;
        });

        // Filter cards and re-append in sorted order
        $.each($cards, function (idx, card) {
            const $card = $(card);
            const cardRating = $card.attr('data-rating');
            const isMatch = activeFilter === 'all' || cardRating === activeFilter;

            if (isMatch) {
                $card.removeClass('d-none');
                visibleCount++;
            } else {
                $card.addClass('d-none');
            }

            $reviewsContainer.append($card);
        });

            const ratingVal = parseInt($ratingInput.val(), 10);

            let hasErrors = false;

            if (isNaN(ratingVal) || ratingVal < 1 || ratingVal > 5) {
                showFeedbackFieldError($ratingInput, 'Пожалуйста, выберите оценку от 1 до 5 звезд.');
                hasErrors = true;
                $ratingInput.focus();
            } else {
                clearFeedbackFieldError($ratingInput);
            }

            if (commentVal.length < 5) {
                showFeedbackFieldError($commentInput, 'Комментарий должен содержать не менее 5 символов.');
                if (!hasErrors) {
                    $commentInput.focus();
                }
                hasErrors = true;
            } else {
                clearFeedbackFieldError($commentInput);
            }

            if (hasErrors) {
                return;
            }

            /*
             * COMMENT ON CLIENT VS SERVER VALIDATION:

/**

function initCallbackForm() {

    if (!$form.length) {
        return;
    }

    const $nameInput = $('#cb-name');
    const $phoneInput = $('#cb-phone');
    const $topicSelect = $('#cb-topic');
    const $messageInput = $('#cb-message');

    $phoneInput.on('input', function () { clearCallbackFieldError($(this)); });

         * event.preventDefault() comment:
         * Prevents the page from refreshing when user submits the callback request.
         * Without preventDefault(): The browser performs default GET/POST, causing a page jump
         * and discarding all user input before confirmation can be shown.

        event.preventDefault();

        if (cleanedDigits.length < 10) {
            showCallbackFieldError($phoneInput, 'Укажите корректный номер телефона (не менее 10 цифр).');
            if (!hasErrors) {
                $phoneInput.focus();
            }
            hasErrors = true;
        } else {
            clearCallbackFieldError($phoneInput);
            $phoneInput.addClass('is-valid');
        }

        if (hasErrors) {
            return;
        }

        /*
         * COMMENT ON CLIENT VS SERVER VALIDATION:
         * Client-side validation gives immediate visual reassurance to the user.
         * However, backend verification is non-negotiable for security:
         * server must enforce rate-limiting (prevent SMS flooding), authenticate users,
         * sanitize against injection attacks, and securely queue callback notifications.
         */
        if ($msgContainer.length) {
            $msgContainer.empty();

            const $alertDiv = $('<div>', {
                class: 'alert alert-success shadow-sm py-3 mt-3 text-start'
            }).hide();

            const $title = $('<strong>', { class: 'd-block mb-1', text: '✅ Запрос успешно зарегистрирован!' });
            const $desc = $('<p>', {
                class: 'mb-1 small',

            });
            const $demoNotice = $('<small>', {
                class: 'text-muted d-block mt-2 border-top pt-1',
                text: '⚠️ Демонстрационный режим: запрос обработан на стороне клиента без отправки на сервер.'

            $alertDiv.slideDown(350, function () {
                $form.trigger('reset');
                $form.find('.is-valid').removeClass('is-valid');
            });
        }
    });

function oldValidateBookingField(fieldName, form) {
    const fullNameInput = form.querySelector('#fullName');
    const phoneInput = form.querySelector('#phoneNumber');
    const dateInput = form.querySelector('#resDate');
    const timeInput = form.querySelector('#resTime');
    const guestsInput = form.querySelector('#guestCount');
    const zoneSelect = form.querySelector('#zone');

    const todayStr = new Date().toISOString().split('T')[0];

    if (fieldName === 'fullName') {
        const val = fullNameInput.value.trim();
        if (val.length === 0) {
            return 'Пожалуйста, укажите ваше имя.';
        }
        if (val.length < 2) {
            return 'Имя должно содержать не менее 2 символов.';
        }
        const nameRegex = /^[A-Za-zА-Яа-яЁёӘәҒғҚқҢңӨөҰұҮүҺһІі\s\-]+$/;
        if (!nameRegex.test(val)) {
            return 'Имя должно состоять только из букв.';
        }
        return null;
    }

    if (fieldName === 'phoneNumber') {
        const val = phoneInput.value.trim();
        const cleanedDigits = val.replace(/\D/g, '');
        if (cleanedDigits.length < 10) {
            return 'Укажите корректный номер телефона (не менее 10 цифр).';
        }
        return null;
    }

    if (fieldName === 'resDate') {
        const val = dateInput.value;
        if (!val) {
            return 'Выберите дату бронирования.';
        }
        if (val < todayStr) {
            return 'Дата бронирования не может быть в прошлом.';
        }
        return null;
    }

    if (fieldName === 'resTime') {
        const val = timeInput.value;
        if (!val) {
            return 'Укажите желаемое время визита.';
        }
        const hour = parseInt(val.split(':')[0], 10);
        if (hour >= 2 && hour < 12) {
            return 'Frau Muller открыт с 12:00 до 02:00 ночи.';
        }
        return null;
    }

    if (fieldName === 'guestCount') {
        const count = parseInt(guestsInput.value, 10);
        if (isNaN(count) || count < 1 || count > 30) {
            return 'Укажите количество гостей от 1 до 30 человек.';
        }
        return null;
    }

    if (fieldName === 'zone') {
        const val = zoneSelect.value;
        const count = parseInt(guestsInput.value, 10);
        if (val === 'vip' && count > 12) {
            return 'VIP-кабина вмещает до 12 гостей. Для больших компаний выберите Основной зал.';
        }
        return null;
    }

    return null;
}

## 3. Three User Journeys (End-to-End Walkthroughs)

Every journey described below can be performed completely and independently by a single visitor using their screen, with no dead ends, no `href="#"`, and with clear completion:

### 🚶 Journey 1: "Найти адрес, режим работы и забронировать столик на вечер"

## 4. Assignment 6: jQuery Migration & Architecture

### A. Подключение jQuery и режим заморозки (Freeze State)
* **Версия библиотеки:** **jQuery 3.7.1 (Official Compressed CDN)**:
  ```html
  <script src="https://code.jquery.com/jquery-3.7.1.min.js"></script>
  ```
* **Порядок загрузки:** Библиотека jQuery подключена строго **до** пользовательских скриптов проекта (`js/main.js`, `js/index.js` и т.д.).
* **Зачем нужен `$(document).ready()` (или `$(function () { ... })`):**
  Обертка `$(document).ready()` гарантирует, что браузер полностью построил DOM-дерево (событие `DOMContentLoaded`), прежде чем скрипты начнут поиск элементов (`$('#booking-form')`, `$('.review-card')`) или привязку событий. Без этого вызовы jQuery к элементам, расположенным ниже по коду, вернули бы пустые коллекции (`length === 0`), приводя к сбоям логики.
* **Соблюдение Freeze:**
  - В HTML-файлах вручную не создавалось никаких новых интерфейсных элементов (все динамические карточки, бейджи, квитанции создаются исключительно через код jQuery).
  - В CSS-файлах не дописывалось никаких новых правил (все изменения внешнего вида осуществляются переключением существующих классов Bootstrap и `base.css`).
  - В разметке полностью отсутствуют инлайн-обработчики (0 атрибутов `onclick`).

---

### B. Три переписанных компонента и файл `old-js-version.js`

Оригинальные версии трех компонентов на чистом JavaScript сохранены в файле [`old-js-version.js`](old-js-version.js), который не подключен ни к одной странице.

#### 1. Компонент 1: Динамический каталог блюд из массива с фильтрами (`js/index.js`)
* **Что заменено:** Множественные вызовы `document.createElement()`, ручные цепочки `appendChild()`, прямое переключение `classList.add()` и разрозненные `addEventListener('click')`.
* **На что в jQuery:** Фабричные конструкторы `$('<div>', { class: '...' })`, цепочки `.append()`, классы `.addClass()`, и делегированный обработчик `$cardsRow.on('click', '.card-dish-item', ...)`.
* **Технический вердикт:** Код стал на 40% короче и значительно нагляднее. Создание разметки из массива объектов через декларативные свойства jQuery исключает громоздкий бойлерплейт создания узлов.

#### 2. Компонент 2: Валидация формы бронирования и калькулятор депозита (`js/reservation.js`)
* **Что заменено:** Вызовы `form.querySelector()`, `input.value.trim()`, `parentElement.appendChild()`, `classList.remove/add`, `form.addEventListener('submit')`.
* **На что в jQuery:** Селекторы `$('#...')`, метод `$.trim()`, обход `.parent()`, `.siblings()`, установка классов `.addClass('is-invalid')`, эффекты `.fadeIn()` для ошибок и `.slideDown()` для подтверждения.
* **Технический вердикт:** Навигация к родительским контейнерам полей и вставка сообщений об ошибках через `.parent().append()` и `.siblings().fadeOut()` гораздо компактнее и надежнее стандартных цепочек DOM.

* **Что заменено:** `parentNode.insertBefore()`, обертка узлов, ручное переключение `input.type = ...`, ручная очистка и обновление `li.className`.
* **На что в jQuery:** Метод `.wrap('<div class="input-group">')`, метод `.prop('type', ...)`, цепочки методов `.removeClass().addClass().text()`, динамическая генерация `$.each()`.
* **Технический вердикт:** Метод `.prop()` гарантирует корректную кроссбраузерную смену типа поля пароля, а цепочки методов лаконично обновляют элементы списка правил.

### D. Изменение страницы: методы работы со значениями и DOM

| Метод | Где применен | Роль в проекте |
|---|---|---|
| `.text()` | Заголовки, бейджи, цены, квитанции | Безопасная установка текста с авто-экранированием HTML (защита от XSS). |
| `.html()` | Иконки SVG (глаз пароля), префиксы | Рендеринг доверенной векторной разметки SVG без создания лишних узлов. |
| `.val()` | Все поля ввода, селекты, даты | Получение и программная установка значений форм (`.val('')`, `$.trim($input.val())`). |
| `.attr()` | `data-*`, `min`, `placeholder`, `title`, `aria-label` | Управление атрибутами HTML-разметки и свойствами доступности. |
| `.prop()` | Чекбоксы, переключатели, `disabled`, `type` | Управление динамическими свойствами браузера (`.prop('checked')`, `.prop('disabled', true)`). |

#### 💡 Разница для защиты проекта:
1. **Разница между `.text()` и `.html()`:**
   - `.text()` считывает или записывает чистый текст, преобразуя специальные символы (`<`, `>`, `&`) в текстовые сущности. Это исключает внедрение вредоносного кода (XSS).
   - `.html()` интерпретирует переданную строку как HTML-код и парсит ее в DOM. Используется только для доверенной разметки (например, SVG-иконки).
2. **Разница между `.attr()` и `.prop()` на чекбоксе:**
   - `.attr('checked')` обращается к исходному атрибуту в HTML-разметке (возвращает строку `"checked"` или `undefined`) и не меняется при кликах пользователя.
   - `.prop('checked')` обращается к живому состоянию DOM-свойства в памяти браузера (возвращает булево `true` или `false`). Для проверки отмечен ли чекбокс пользователем, необходимо использовать `.prop('checked')`.

### E. События и делегирование (Event Delegation)

Все обработчики событий привязаны через единый метод `.on()`.

#### 1. Типы событий в проекте:
* `click`: клики по кнопкам добавления в корзину, фильтрам, очистке, карточкам, кнопке «наверх».
* `submit`: отправка форм бронирования (`reservation.html`), доставки (`order.html`), отзывов (`feedback.html`), обратного звонка (`contacts.html`), авторизации/регистрации (`main.js`).
* `input`: живой поиск по блюдам, динамический подсчет стоимости при изменении порций, проверка надежности пароля на лету.
* `change`: переключение даты визита, выбор зала, переключение радио-кнопок способа доставки, оплаты, повода.
* `scroll`: отслеживание прокрутки страницы для плавного появления плавающей кнопки WhatsApp и возврата наверх.

#### 2. Делегированные обработчики:
```javascript
// index.js — делегирование на контейнер карточек:
$cardsRow.on('click', '.card-dish-item', function (event) { ... });

// order.js — делегирование на контейнер корзины:
$cartContainer.on('click', '.btn-remove-cart-item', function (event) { ... });

// menu.js — делегирование на таблицу:
$('table.table').on('click', '.btn-stepper-plus', function (event) { ... });
```

#### ❓ Почему прямое связывание (`$('.card-dish-item').on('click', ...)`) не сработало бы:
Прямой вызов ищет элементы `.card-dish-item` один раз в момент выполнения строки скрипта и навешивает слушатели только на те узлы, которые уже присутствуют в DOM. Когда каталог фильтруется или перерисовывается (с очисткой контейнера `.empty()`), старые узлы удаляются, а новые создаются заново. На вновь созданных карточках никаких слушателей уже не будет — кнопки окажутся «мертвыми».
Делегированный обработчик вешается на постоянного предка (`$cardsRow`), который всегда присутствует на странице. Благодаря механизму всплытия событий (Event Bubbling), клик на новой карточке поднимается вверх до предка, который сверяет селектор `.card-dish-item` и выполняет код.

#### ❓ Зачем нужен `event.preventDefault()`:
Без `event.preventDefault()` форма выполняет стандартное браузерное действие — отправку HTTP-запроса с немедленной перезагрузкой страницы. В результате введенные данные сбрасываются, страница моргает, а сообщения валидации и квитанции об успехе исчезают еще до того, как пользователь успеет их прочитать.

В проекте задействовано 5 различных эффектов jQuery, каждый из которых служит конкретному сценарию пользователя:

### G. Валидация форм на jQuery

Формы бронирования стола (`reservation.html`), доставки (`order.html`), отзывов (`feedback.html`) и обратной связи (`contacts.html`) переведены на строгую клиентскую валидацию:
1. Отмена стандартной отправки через `event.preventDefault()`.
2. Предварительная очистка пробелов по краям через `$.trim()`.
3. Индивидуальная подсветка каждого невалидного поля классом `.is-invalid` и вывод персонального сообщения об ошибке в `.invalid-feedback` **непосредственно рядом с полем**, а не одним общим сообщением на всю форму.
4. Фокусировка на первом ошибочном поле (`$firstInvalid.focus()`).
5. Мгновенная очистка ошибки при исправлении поля пользователем через обработчики событий `.on('input change')`.
6. Сброс формы (`$form.trigger('reset')`) и плавный показ квитанции только при 100% прохождении всех правил.

#### 🛡️ Почему проверка в браузере никогда не заменяет проверку на сервере:
1. Клиентская валидация выполняется на стороне пользователя. Ее можно полностью обойти: отключить JavaScript, изменить код через DevTools, или отправить прямой HTTP-запрос через curl/Postman.
2. Только сервер имеет доступ к базе данных и может проверить реальные ограничения: занятость столиков, наличие ингредиентов, уникальность email или подлинность платежного токена.
3. Клиентская проверка создана исключительно для удобства пользователя (UX) — мгновенно подсказать формат телефона или обязательность поля до отправки запроса. Безопасность и целостность данных всегда гарантирует только сервер.

## 5. Вопросы и практические сценарии к защите проекта (Defense Guide)

### 💬 Ответы на контрольные вопросы преподавателя:
1. **«Что делает эта строка jQuery и каков ее эквивалент на чистом JS?»**
   - jQuery: `$('#dishSelect').val();`
   - Чистый JS: `document.getElementById('dishSelect').value;`
   - jQuery: `$(this).closest('.card').find('.card-title').addClass('text-danger');`
   - Чистый JS: `this.closest('.card').querySelector('.card-title').classList.add('text-danger');`
2. **«Почему делегированный обработчик перехватывает элементы, которых еще не было в DOM?»**
   - Обработчик вешается не на сами элементы, а на их статического родителя (`#container`). Событие клика на новом элементе всплывает (event bubbling) по дереву вверх до предка. Предок перехватывает клик, проверяет селектор в `event.target` и исполняет функцию.
3. **«В чем разница между `.text()` и `.html()`?»**
   - `.text()` экранирует спецсимволы и работает с чистым текстом (защита от XSS). `.html()` парсит строку в DOM-элементы.
4. **«В чем разница между `.attr()` и `.prop()` на чекбоксе?»**
   - `.attr('checked')` возвращает состояние атрибута из исходной HTML-разметки. `.prop('checked')` возвращает текущее динамическое булево состояние (`true`/`false`).
5. **«Почему клиентская валидация не заменяет серверную?»**
   - Клиентский код можно отключить, модифицировать или обойти прямым POST-запросом. Сервер — единственная доверенная среда для обеспечения безопасности и проверки базы данных.

## 1. Questions Asked & Concepts Researched

### Topic 1: Plain JavaScript Event Delegation vs Direct Attachment
* **Question:** Why is a delegated event listener (`parent.addEventListener('click', ...)`) preferred over attaching handlers to individual items created dynamically?
* **Explanation Received:** Elements created after the initial DOM load (such as dynamically appended order items or receipt cards) do not have listeners directly attached to them. By attaching the handler to a stable parent element and using `e.target` or `e.target.closest()`, any current or future child elements are caught through event bubbling (DOM Level 3 event propagation model).
* **Application in Project:** Applied in `js/order.js` on `#order-form` to catch clicks on reset buttons, order receipt dismissal, and quick actions.

### Topic 2: Strict Separation of Plain JavaScript and jQuery
* **Question:** What is the technical penalty for mixing jQuery and Plain JS in the same file or feature?
* **Explanation Received:** Mixing APIs (e.g. using `document.getElementById` inside a jQuery chain or wrapping raw DOM objects unnecessarily) violates modular design, creates maintenance confusion, and breaks the course requirement. Features must be strictly decoupled: `js/order.js`, `js/reservation.js`, and `js/auth.js` use pure Plain JavaScript, while `js/feedback.js` and `js/menu.js` use jQuery.

### Topic 3: jQuery Traversal vs Global ID Lookups

### Log Entry 1: jQuery Loading Lifecycle and `$(document).ready()` Mechanics
* **Date:** 2026-10-28
* **Topic / Inquiry:**
  > "Why is `$(document).ready()` (or `$(function() { ... })`) required when scripts are placed at the bottom of the page, and what specific browser events does it hook into compared to `window.onload`?"
* **Technical Clarification Received:**

Wrapping code in `$(document).ready()` guarantees that jQuery element selectors (such as `$('#booking-form')` or `$('.review-card')`) will find valid DOM nodes and never return empty collections.

### Log Entry 2: Direct Binding vs. Event Delegation (`.on()`)
* **Date:** 2026-10-29
* **Topic / Inquiry:**
  > "Why does binding a click listener directly via `$('.item').on('click', handler)` fail for dynamic cards generated in code, and how does `$('#container').on('click', '.item', handler)` resolve this?"
* **Technical Clarification Received:**
  Direct binding evaluates `$('.item')` once when the statement executes. It traverses the DOM at that precise moment and attaches event listeners only to elements currently in the tree. When a list or catalog is filtered, sorted, or re-rendered with `.empty()` and `.append()`, the old elements are destroyed and brand-new DOM elements are created with zero attached listeners.
  Delegated event handling attaches a single listener to a persistent ancestor (e.g. `$('#cardsRow')` or `$('#cartContainer')`). Due to DOM event bubbling, when a child `.card-dish-item` is clicked, the event bubbles up the ancestor tree. The parent container checks if `event.target` matches the delegate selector (`.card-dish-item`) and executes the handler with `$(this)` bound to the matching child.
* **Project Implementation:**
  - `js/index.js`: `$cardsRow.on('click', '.card-dish-item', ...)`
  - `js/order.js`: `$cartContainer.on('click', '.btn-remove-cart-item', ...)`
  - `js/menu.js`: `$('table.table').on('click', '.btn-stepper-plus', ...)`
  - `js/menu.js`: `$modalItemsContainer.on('click', '.btn-modal-plus', ...)`

### Log Entry 3: `.attr()` vs. `.prop()` and `.text()` vs. `.html()`
* **Date:** 2026-10-29
* **Topic / Inquiry:**
  > "What is the critical semantic and runtime difference between `.attr()` and `.prop()` on checkboxes/inputs, and why does `.text()` prevent XSS while `.html()` should be reserved for trusted markup?"
* **Technical Clarification Received:**
  - `.attr('checked')` queries the static HTML markup attribute string (`"checked"` or `undefined`). It does not change when the user clicks the checkbox on the screen.
  - `.prop('checked')` queries the live, dynamic DOM property maintained by the browser engine (`true` or `false`). For runtime user interactions, `.prop()` is strictly required.
  - `.text(val)` internally invokes `textContent` (or `innerText`), automatically escaping `< > & "` characters into text nodes, rendering malicious script tags harmless.
  - `.html(val)` invokes `innerHTML`, parsing strings into executable DOM elements. It is appropriate only for safe developer-authored icons (e.g., SVG eye toggles), never for raw user input.
* **Project Implementation:**
  Applied on `reservation.html` (`#wantsNotification.prop('checked')`), `order.html` (`#extraSauce.prop('checked')`, input `.prop('disabled', true)`), and throughout text updates.

### Log Entry 5: Client-Side vs. Server-Side Validation Philosophy
* **Date:** 2026-10-30
* **Topic / Inquiry:**
  > "Why does rigorous client-side form validation (regex, required, real-time error clearing) never substitute for server-side validation?"
* **Technical Clarification Received:**
  Client-side validation is solely an interface optimization designed for user convenience (reducing round-trips and giving immediate visual hints). It is never a security mechanism because:
  1. The client controls the execution environment: scripts can be disabled, modified via browser DevTools, or bypassed using automated HTTP clients (curl, Postman).
  2. Business constraints (table capacity, unique database constraints, live inventory) can only be verified against persistent server records.
  3. Backend validation is the only true barrier preventing injection attacks, data corruption, and unauthorized state changes.
* **Project Implementation:**
  Extensively documented in `js/reservation.js`, `js/order.js`, and `README.md`.
